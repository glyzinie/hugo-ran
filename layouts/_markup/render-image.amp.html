{{- $page := .Page -}} {{- with .PageInner -}}{{- $page = . -}}{{- end -}} {{- $attributes := .Attributes | default (dict) -}} {{- $explicitWidth := index $attributes "width" | default "" -}} {{- $explicitHeight := index $attributes "height" | default "" -}} {{- $hasExplicitDimensions := and (findRE `^[1-9][0-9]*$` (printf "%v" $explicitWidth)) (findRE `^[1-9][0-9]*$` (printf "%v" $explicitHeight)) -}} {{- $probeRemote := true -}} {{- if or $hasExplicitDimensions (and (isset $page.Site.Params "remoteimagedimensions") (eq (index $page.Site.Params "remoteimagedimensions") false)) (and (isset $page.Params "remoteimagedimensions") (eq (index $page.Params "remoteimagedimensions") false)) -}} {{- $probeRemote = false -}} {{- end -}} {{- $image := partial "images/resolve.html" (dict "page" $page "destination" .Destination "probeRemote" $probeRemote) -}} {{- $width := $image.width -}} {{- $height := $image.height -}} {{- /* Authors can provide real dimensions for files Hugo cannot inspect, such as remote images and SVGs. */ -}} {{- if $hasExplicitDimensions -}} {{- $width = int $explicitWidth -}} {{- $height = int $explicitHeight -}} {{- end -}} {{- if and (gt $width 0) (gt $height 0) -}} {{- else -}} {{- $label := .PlainText | default .Title | default $image.src -}} {{ $label }} {{- end -}}
